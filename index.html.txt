        function selectGame(gameId, element) {

            if (activeCatalog) {
                activeCatalog.classList.add('active');
            }

            document.getElementById('order-form-box').style.display = 'none';
            document.querySelectorAll('.games-grid, .catalog-section, .section-title').forEach(el => el.style.display = 'none');
            document.getElementById('checkout-screen').style.display = 'block';

            document.getElementById('summary-package').innerText = '🔥 ' + selectedItemName + ' - $ ' + selectedItemPrice + ' COP';
            document.getElementById('summary-id').innerText = 'ID de Jugador: ' + id;
            updateCheckoutUI(1); // Paso inicial

            if (chosenPayment === 'Nequi') {
                document.getElementById('payment-title').innerText = '📱 Datos para pagar por Nequi';
                document.getElementById('account-number').innerText = '3136921834';
            } else {
                document.getElementById('payment-title').innerText = '🏦 Datos para pagar por Bancolombia (Ahorros)';
                document.getElementById('account-number').innerText = '48176374686';
            }
        }

        function copyAccountNumber() {
            const accNum = document.getElementById('account-number').innerText;
            navigator.clipboard.writeText(accNum);
            alert('¡Número ' + accNum + ' copiado al portapapeles! 📋✨');
        }

        function submitFinalOrder() {
            const id = document.getElementById('playerId').value;
            const clientName = document.getElementById('clientName').value || "Cliente Anónimo";
            const timeNow = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            const orderKey = 'ord_' + Date.now();

            const newOrder = {
                key: orderKey,
                id: id,
                name: clientName,
                item: selectedItemName,
                price: selectedItemPrice,
                method: chosenPayment,
                time: timeNow,
                step: 1 // 1: Pedido, 2: Revisando, 3: Recargando, 4: Enviado
            };

            let orders = JSON.parse(localStorage.getItem('asun_orders') || '[]');
            orders.unshift(newOrder);
            localStorage.setItem('asun_orders', JSON.stringify(orders));

            currentActiveOrderKey = orderKey;
            alert('¡Comprobante enviado con éxito! 💖 Tu pedido fue registrado.');
            
            // Dejar al cliente viendo su pasarela con estado 1 actualizándose solo
            document.getElementById('clientName').value = '';
        }

        function updateCheckoutUI(step) {
            // Actualizar circulitos con corazones
            const c1 = document.getElementById('circle-1');
            const c2 = document.getElementById('circle-2');
            const c3 = document.getElementById('circle-3');
            const c4 = document.getElementById('circle-4');
            const statusText = document.getElementById('live-status-text');

            c1.className = "step-circle active";
            c2.className = "step-circle";
            c3.className = "step-circle";
            c4.className = "step-circle";

            if(step >= 2) { c2.className = "step-circle active"; c2.innerText = "💖"; } else { c2.innerText = "2"; }
            if(step >= 3) { c3.className = "step-circle active"; c3.innerText = "💖"; } else { c3.innerText = "3"; }
            if(step >= 4) { c4.className = "step-circle active"; c4.innerText = "💖"; } else { c4.innerText = "4"; }

            const texts = ["", "Estado: 💖 Pedido recibido", "Estado: 👀 Revisando pago", "Estado: ⚡ Recargando diamantes", "Estado: ✨ ¡Diamantes enviados con éxito!"];
            statusText.innerText = texts[step] || "";
        }

        function setOrderStep(orderKey, newStep) {
            let orders = JSON.parse(localStorage.getItem('asun_orders') || '[]');
            orders = orders.map(ord => {
                if(ord.key === orderKey) {
                    ord.step = newStep;
                }
                return ord;
            });
            localStorage.setItem('asun_orders', JSON.stringify(orders));
            renderOrders();

            // Si el cliente tiene este pedido abierto en pantalla, se le actualiza
            if(currentActiveOrderKey === orderKey) {
                updateCheckoutUI(newStep);
            }
        }

        function openAdminPanel() {
            document.getElementById('store-main-view').style.display = 'none';
            document.getElementById('checkout-screen').style.display = 'none';
            document.getElementById('admin-screen').style.display = 'block';
            renderOrders();
        }

            document.getElementById('admin-screen').style.display = 'none';
            document.getElementById('store-main-view').style.display = 'block';
            document.querySelectorAll('.games-grid, .section-title').forEach(el => el.style.display = 'grid');
            document.querySelectorAll('.catalog-section').forEach(el => el.classList.remove('active'));
            document.getElementById('catalog-freefire').classList.add('active');
            document.getElementById('order-form-box').style.display = 'block';
        }

        function renderOrders() {
            const listContainer = document.getElementById('admin-orders-list');
            const orders = JSON.parse(localStorage.getItem('asun_orders') || '[]');

            if (orders.length === 0) {
                listContainer.innerHTML = '<p style="color: var(--text-muted); text-align: center; font-size: 0.8rem; padding: 20px;">No hay pedidos activos. ¡A facturar se dijo! 🚀</p>';
                return;
            }

            let html = '';
            orders.forEach((order) => {
                const statusNames = ["", "Pendiente", "Revisando Pago", "Recargando", "Diamantes Enviados"];
                html += `
                    <div class="order-card-admin">
                        <div style="display:flex; justify-content:space-between; margin-bottom:5px;">
                            <span style="color:#d8b4fe; font-weight:bold; font-size:0.85rem;">📦 ${order.item}</span>
                            <span style="color:#38bdf8; font-size:0.8rem;">$ ${order.price} COP</span>
                        </div>
                        <p style="font-size:0.75rem; color:var(--text-muted);">👤 Titular: <b>${order.name}</b></p>
                        <p style="font-size:0.75rem; color:var(--text-muted);">🎮 ID: <b style="color:white;">${order.id}</b> | 💳 ${order.method}</p>
                        <p style="font-size:0.75rem; color:#fde047; margin-top:4px;">Estado actual: <b>${statusNames[order.step]}</b></p>
                        
                        <div class="admin-controls">
                            <button class="admin-action-btn" onclick="setOrderStep('${order.key}', 2)">👀 Revisando</button>
                            <button class="admin-action-btn" onclick="setOrderStep('${order.key}', 3)">⚡ Recargando</button>
                            <button class="admin-action-btn" style="background:#7c3aed;" onclick="setOrderStep('${order.key}', 4)">✨ Enviados</button>
                        </div>
                    </div>
                `;
            });
            listContainer.innerHTML = html;
        }

        function clearOrders() {
            if(confirm('¿Seguro que quieres limpiar todo el historial?')) {
                localStorage.removeItem('asun_orders');
                renderOrders();
            }
        }
